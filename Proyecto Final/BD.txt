-- 1. Crear la base de datos si no existe
CREATE DATABASE IF NOT EXISTS aerolineas_db;
USE aerolineas_db;

-- 2. Crear la tabla de usuarios
CREATE TABLE IF NOT EXISTS usuarios (
  id INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(100),
  email VARCHAR(100),
  dni VARCHAR(8),
  password VARCHAR(100),
  rol VARCHAR(20)
);

-- 3. Crear la tabla de vuelos (con precio integrado)
CREATE TABLE IF NOT EXISTS vuelos (
  id INT AUTO_INCREMENT PRIMARY KEY,
  origen VARCHAR(100),
  destino VARCHAR(100),
  ida VARCHAR(50),
  vuelta VARCHAR(50),
  pasajeros INT,
  codigo VARCHAR(50),
  precio VARCHAR(50)
);
